import { z } from 'zod';

export const createQuizSchema = z
  .object({
    courseId: z.string().uuid('Invalid course ID'),
    title: z.string().trim().min(3, 'Title must be at least 3 characters'),
    description: z
      .string()
      .trim()
      .min(5, 'Description must be at least 5 characters'),
    durationMinutes: z.number().int().positive('Duration must be positive'),
    maxMarks: z.number().positive('Maximum marks must be positive'),
    maxAttempts: z.number().int().positive().default(1),
    startsAt: z.iso.datetime().optional(),
    endsAt: z.iso.datetime().optional(),
  })
  .refine(
    (quiz) =>
      !quiz.startsAt ||
      !quiz.endsAt ||
      new Date(quiz.startsAt) < new Date(quiz.endsAt),
    {
      message: 'Quiz end time must be after start time',
      path: ['endsAt'],
    },
  );

export const updateQuizSchema = z
  .object({
    title: z
      .string()
      .trim()
      .min(3, 'Title must be at least 3 characters')
      .optional(),
    description: z
      .string()
      .trim()
      .min(5, 'Description must be at least 5 characters')
      .optional(),
    durationMinutes: z.number().int().positive().optional(),
    maxMarks: z.number().positive().optional(),
    maxAttempts: z.number().int().positive().optional(),
    startsAt: z.iso.datetime().nullable().optional(),
    endsAt: z.iso.datetime().nullable().optional(),
    showMarks: z.boolean().optional(),
    showCorrectAnswers: z.boolean().optional(),
  })
  .refine(
    (quiz) =>
      !quiz.startsAt ||
      !quiz.endsAt ||
      new Date(quiz.startsAt) < new Date(quiz.endsAt),
    {
      message: 'Quiz end time must be after start time',
      path: ['endsAt'],
    },
  );

export const releaseQuizResultsSchema = z
  .object({
    showMarks: z.boolean(),
    showCorrectAnswers: z.boolean(),
  })
  .strict();

export const createQuizQuestionSchema = z.object({
  questionText: z
    .string()
    .trim()
    .min(5, 'Question must be at least 5 characters'),
  optionA: z.string().trim().min(1),
  optionB: z.string().trim().min(1),
  optionC: z.string().trim().min(1),
  optionD: z.string().trim().min(1),
  correctOption: z.enum(['A', 'B', 'C', 'D']),
  marks: z.number().positive().default(1),
});

export const submitQuizAnswerSchema = z.object({
  questionId: z.string().uuid('Invalid question ID'),
  selectedOption: z.enum(['A', 'B', 'C', 'D']),
});

import { Prisma } from '../../generated/prisma/client.js';
import { PrismaClient } from '../../generated/prisma/client.js';

import type {
  Quiz,
  QuizRepository,
  ReleaseQuizResultsInput,
} from './quiz.repository.interface.js';

import type { CreateQuizInput, UpdateQuizInput } from './quiz.schema.js';

export class PrismaQuizRepository implements QuizRepository {
  constructor(private readonly prisma: PrismaClient) {}

  async create(input: CreateQuizInput): Promise<Quiz> {
    const quiz = await this.prisma.quiz.create({
      data: {
        courseId: input.courseId,
        title: input.title,
        description: input.description,
        durationMinutes: input.durationMinutes,
        maxMarks: input.maxMarks,
        maxAttempts: input.maxAttempts,
        ...(input.startsAt && {
          startsAt: new Date(input.startsAt),
        }),
        ...(input.endsAt && {
          endsAt: new Date(input.endsAt),
        }),
      },
    });

    return this.toQuiz(quiz);
  }

  async findAllByCourseId(courseId: string): Promise<Quiz[]> {
    const quizzes = await this.prisma.quiz.findMany({
      where: { courseId },
      orderBy: { createdAt: 'desc' },
    });

    return quizzes.map((quiz) => this.toQuiz(quiz));
  }

  async findById(id: string): Promise<Quiz | undefined> {
    const quiz = await this.prisma.quiz.findUnique({
      where: { id },
    });

    return quiz ? this.toQuiz(quiz) : undefined;
  }

  async update(id: string, input: UpdateQuizInput): Promise<Quiz | undefined> {
    try {
      const quiz = await this.prisma.quiz.update({
        where: { id },
        data: {
          ...(input.title !== undefined && {
            title: input.title,
          }),
          ...(input.description !== undefined && {
            description: input.description,
          }),
          ...(input.durationMinutes !== undefined && {
            durationMinutes: input.durationMinutes,
          }),
          ...(input.maxMarks !== undefined && {
            maxMarks: input.maxMarks,
          }),
          ...(input.maxAttempts !== undefined && {
            maxAttempts: input.maxAttempts,
          }),
          ...(input.startsAt !== undefined && {
            startsAt: input.startsAt ? new Date(input.startsAt) : null,
          }),
          ...(input.endsAt !== undefined && {
            endsAt: input.endsAt ? new Date(input.endsAt) : null,
          }),
          ...(input.showMarks !== undefined && {
            showMarks: input.showMarks,
          }),
          ...(input.showCorrectAnswers !== undefined && {
            showCorrectAnswers: input.showCorrectAnswers,
          }),
        },
      });

      return this.toQuiz(quiz);
    } catch (error: unknown) {
      if (this.isRecordNotFoundError(error)) {
        return undefined;
      }

      throw error;
    }
  }

  async countQuestions(id: string): Promise<number> {
    return this.prisma.quizQuestion.count({
      where: { quizId: id },
    });
  }

  async publish(id: string): Promise<Quiz | undefined> {
    try {
      const quiz = await this.prisma.quiz.update({
        where: { id },
        data: {
          status: 'published',
        },
      });

      return this.toQuiz(quiz);
    } catch (error: unknown) {
      if (this.isRecordNotFoundError(error)) {
        return undefined;
      }

      throw error;
    }
  }

  async releaseResults(
    id: string,
    input: ReleaseQuizResultsInput,
  ): Promise<Quiz | undefined> {
    try {
      const quiz = await this.prisma.quiz.update({
        where: { id },
        data: {
          resultsReleasedAt: new Date(),
          showMarks: input.showMarks,
          showCorrectAnswers: input.showCorrectAnswers,
        },
      });

      return this.toQuiz(quiz);
    } catch (error: unknown) {
      if (this.isRecordNotFoundError(error)) {
        return undefined;
      }

      throw error;
    }
  }

  async delete(id: string): Promise<boolean> {
    try {
      await this.prisma.quiz.delete({
        where: { id },
      });

      return true;
    } catch (error: unknown) {
      if (this.isRecordNotFoundError(error)) {
        return false;
      }

      throw error;
    }
  }

  private isRecordNotFoundError(error: unknown): boolean {
    return (
      error instanceof Prisma.PrismaClientKnownRequestError &&
      error.code === 'P2025'
    );
  }

  private toQuiz(quiz: {
    id: string;
    courseId: string;
    title: string;
    description: string;
    durationMinutes: number;
    maxMarks: number;
    maxAttempts: number;
    status: Quiz['status'];
    startsAt: Date | null;
    endsAt: Date | null;
    resultsReleasedAt: Date | null;
    showMarks: boolean;
    showCorrectAnswers: boolean;
    createdAt: Date;
    updatedAt: Date;
  }): Quiz {
    return {
      id: quiz.id,
      courseId: quiz.courseId,
      title: quiz.title,
      description: quiz.description,
      durationMinutes: quiz.durationMinutes,
      maxMarks: quiz.maxMarks,
      maxAttempts: quiz.maxAttempts,
      status: quiz.status,
      startsAt: quiz.startsAt,
      endsAt: quiz.endsAt,
      resultsReleasedAt: quiz.resultsReleasedAt,
      showMarks: quiz.showMarks,
      showCorrectAnswers: quiz.showCorrectAnswers,
      createdAt: quiz.createdAt,
      updatedAt: quiz.updatedAt,
    };
  }
}

import { randomUUID } from 'node:crypto';

import type { CreateQuizInput, UpdateQuizInput } from './quiz.schema.js';
import type {
  Quiz,
  QuizRepository,
  ReleaseQuizResultsInput,
} from './quiz.repository.interface.js';

export class InMemoryQuizRepository implements QuizRepository {
  private readonly quizzes = new Map<string, Quiz>();
  private readonly questionCounts = new Map<string, number>();

  async create(input: CreateQuizInput): Promise<Quiz> {
    const now = new Date();

    const quiz: Quiz = {
      id: randomUUID(),
      courseId: input.courseId,
      title: input.title,
      description: input.description,
      durationMinutes: input.durationMinutes,
      maxMarks: input.maxMarks,
      maxAttempts: input.maxAttempts,
      status: 'draft',
      startsAt: input.startsAt ? new Date(input.startsAt) : null,
      endsAt: input.endsAt ? new Date(input.endsAt) : null,
      resultsReleasedAt: null,
      showMarks: false,
      showCorrectAnswers: false,
      createdAt: now,
      updatedAt: now,
    };

    this.quizzes.set(quiz.id, quiz);
    this.questionCounts.set(quiz.id, 0);

    return { ...quiz };
  }

  async findAllByCourseId(courseId: string): Promise<Quiz[]> {
    return Array.from(this.quizzes.values())
      .filter((quiz) => quiz.courseId === courseId)
      .map((quiz) => ({ ...quiz }));
  }

  async findById(id: string): Promise<Quiz | undefined> {
    const quiz = this.quizzes.get(id);
    return quiz ? { ...quiz } : undefined;
  }

  async update(
    id: string,
    input: UpdateQuizInput,
  ): Promise<Quiz | undefined> {
    const existingQuiz = this.quizzes.get(id);

    if (!existingQuiz) {
      return undefined;
    }

    const updatedQuiz: Quiz = {
      ...existingQuiz,
      title: input.title ?? existingQuiz.title,
      description: input.description ?? existingQuiz.description,
      durationMinutes: input.durationMinutes ?? existingQuiz.durationMinutes,
      maxMarks: input.maxMarks ?? existingQuiz.maxMarks,
      maxAttempts: input.maxAttempts ?? existingQuiz.maxAttempts,
      startsAt:
        input.startsAt === undefined
          ? existingQuiz.startsAt
          : input.startsAt === null
            ? null
            : new Date(input.startsAt),
      endsAt:
        input.endsAt === undefined
          ? existingQuiz.endsAt
          : input.endsAt === null
            ? null
            : new Date(input.endsAt),
      showMarks: input.showMarks ?? existingQuiz.showMarks,
      showCorrectAnswers:
        input.showCorrectAnswers ?? existingQuiz.showCorrectAnswers,
      updatedAt: new Date(),
    };

    this.quizzes.set(id, updatedQuiz);
    return { ...updatedQuiz };
  }

  async countQuestions(id: string): Promise<number> {
    return this.questionCounts.get(id) ?? 0;
  }

  async publish(id: string): Promise<Quiz | undefined> {
    const existingQuiz = this.quizzes.get(id);

    if (!existingQuiz) {
      return undefined;
    }

    const publishedQuiz: Quiz = {
      ...existingQuiz,
      status: 'published',
      updatedAt: new Date(),
    };

    this.quizzes.set(id, publishedQuiz);
    return { ...publishedQuiz };
  }

  async releaseResults(
    id: string,
    input: ReleaseQuizResultsInput,
  ): Promise<Quiz | undefined> {
    const existingQuiz = this.quizzes.get(id);

    if (!existingQuiz) {
      return undefined;
    }

    const now = new Date();

    const updatedQuiz: Quiz = {
      ...existingQuiz,
      resultsReleasedAt: now,
      showMarks: input.showMarks,
      showCorrectAnswers: input.showCorrectAnswers,
      updatedAt: now,
    };

    this.quizzes.set(id, updatedQuiz);
    return { ...updatedQuiz };
  }

  async delete(id: string): Promise<boolean> {
    this.questionCounts.delete(id);
    return this.quizzes.delete(id);
  }
}

import { AppError } from '../../errors/app-error.js';
import type { CourseRepository } from '../courses/course.repository.interface.js';

import type {
  CreateQuizInput,
  UpdateQuizInput,
  ReleaseQuizResultsInput,
} from './quiz.schema.js';
import type { Quiz, QuizRepository } from './quiz.repository.interface.js';

type QuizReader = {
  role: 'admin' | 'faculty' | 'student';
};

export class QuizService {
  constructor(
    private readonly quizRepository: QuizRepository,
    private readonly courseRepository: CourseRepository,
  ) {}

  async create(input: CreateQuizInput): Promise<Quiz> {
    const course = await this.courseRepository.findById(input.courseId);

    if (!course) {
      throw new AppError('Course not found', 404, 'COURSE_NOT_FOUND');
    }

    return this.quizRepository.create(input);
  }

  async getAllByCourseId(courseId: string, user: QuizReader): Promise<Quiz[]> {
    const course = await this.courseRepository.findById(courseId);

    if (!course) {
      throw new AppError('Course not found', 404, 'COURSE_NOT_FOUND');
    }

    const quizzes = await this.quizRepository.findAllByCourseId(courseId);

    if (user.role !== 'student') {
      return quizzes;
    }

    return quizzes
      .filter((quiz) => quiz.status === 'published')
      .map((quiz) => this.forStudent(quiz));
  }

  async getById(id: string, user: QuizReader): Promise<Quiz> {
    const quiz = await this.quizRepository.findById(id);

    if (!quiz) {
      throw new AppError('Quiz not found', 404, 'QUIZ_NOT_FOUND');
    }

    if (user.role === 'student' && quiz.status !== 'published') {
      throw new AppError('Quiz not found', 404, 'QUIZ_NOT_FOUND');
    }

    return user.role === 'student' ? this.forStudent(quiz) : quiz;
  }

  async update(id: string, input: UpdateQuizInput): Promise<Quiz> {
    const quiz = await this.quizRepository.update(id, input);

    if (!quiz) {
      throw new AppError('Quiz not found', 404, 'QUIZ_NOT_FOUND');
    }

    return quiz;
  }

  async publish(id: string): Promise<Quiz> {
    const quiz = await this.quizRepository.findById(id);

    if (!quiz) {
      throw new AppError('Quiz not found', 404, 'QUIZ_NOT_FOUND');
    }

    const questionCount = await this.quizRepository.countQuestions(id);

    if (questionCount === 0) {
      throw new AppError(
        'Cannot publish a quiz without questions',
        400,
        'QUIZ_HAS_NO_QUESTIONS',
      );
    }

    const publishedQuiz = await this.quizRepository.publish(id);

    if (!publishedQuiz) {
      throw new AppError('Quiz not found', 404, 'QUIZ_NOT_FOUND');
    }

    return publishedQuiz;
  }

  async releaseResults(
    id: string,
    input: ReleaseQuizResultsInput,
  ): Promise<Quiz> {
    const quiz = await this.quizRepository.releaseResults(id, input);

    if (!quiz) {
      throw new AppError('Quiz not found', 404, 'QUIZ_NOT_FOUND');
    }

    return quiz;
  }

  async delete(id: string): Promise<void> {
    const deleted = await this.quizRepository.delete(id);

    if (!deleted) {
      throw new AppError('Quiz not found', 404, 'QUIZ_NOT_FOUND');
    }
  }

  private forStudent(quiz: Quiz): Quiz {
    return {
      ...quiz,
      showMarks: quiz.resultsReleasedAt !== null && quiz.showMarks,
      showCorrectAnswers:
        quiz.resultsReleasedAt !== null && quiz.showCorrectAnswers,
    };
  }
}

import type { FastifyInstance } from 'fastify';

import { AppError } from '../../errors/app-error.js';
import { authenticate } from '../../middleware/authenticate.js';
import { authorize } from '../../middleware/authorize.js';

import {
  createQuizSchema,
  updateQuizSchema,
  releaseQuizResultsSchema,
} from './quiz.schema.js';

import type { QuizService } from './quiz.service.js';

type QuizRouteUser = {
  role: 'admin' | 'faculty' | 'student';
};

export async function quizRoutes(
  app: FastifyInstance,
  service: QuizService,
): Promise<void> {
  // Create a quiz
  app.post(
    '/quizzes',
    {
      preHandler: [authenticate, authorize('faculty', 'admin')],
    },
    async (request, reply) => {
      const parsed = createQuizSchema.safeParse(request.body);

      if (!parsed.success) {
        throw new AppError('Invalid quiz data', 400, 'VALIDATION_ERROR');
      }

      const quiz = await service.create(parsed.data);

      return reply.code(201).send({
        success: true,
        data: quiz,
      });
    },
  );

  // Publish a quiz: Faculty and Admin only
  app.post(
    '/quizzes/:id/publish',
    {
      preHandler: [authenticate, authorize('faculty', 'admin')],
    },
    async (request, reply) => {
      const { id } = request.params as { id: string };

      if (!isUuid(id)) {
        throw new AppError('Invalid quiz ID', 400, 'VALIDATION_ERROR');
      }

      const quiz = await service.publish(id);

      return reply.code(200).send({
        success: true,
        data: quiz,
      });
    },
  );

  // Get quizzes for a course
  app.get(
    '/courses/:courseId/quizzes',
    {
      preHandler: [authenticate],
    },
    async (request, reply) => {
      const { courseId } = request.params as { courseId: string };

      if (!isUuid(courseId)) {
        throw new AppError('Invalid course ID', 400, 'VALIDATION_ERROR');
      }

      const user = request.user as QuizRouteUser;
      const quizzes = await service.getAllByCourseId(courseId, user);

      return reply.code(200).send({
        success: true,
        data: quizzes,
      });
    },
  );

  // Get a quiz by ID
  app.get(
    '/quizzes/:id',
    {
      preHandler: [authenticate],
    },
    async (request, reply) => {
      const { id } = request.params as { id: string };

      if (!isUuid(id)) {
        throw new AppError('Invalid quiz ID', 400, 'VALIDATION_ERROR');
      }

      const user = request.user as QuizRouteUser;
      const quiz = await service.getById(id, user);

      return reply.code(200).send({
        success: true,
        data: quiz,
      });
    },
  );

  // Update a quiz
  app.patch(
    '/quizzes/:id',
    {
      preHandler: [authenticate, authorize('faculty', 'admin')],
    },
    async (request, reply) => {
      const { id } = request.params as { id: string };

      if (!isUuid(id)) {
        throw new AppError('Invalid quiz ID', 400, 'VALIDATION_ERROR');
      }

      const parsed = updateQuizSchema.safeParse(request.body);

      if (!parsed.success) {
        throw new AppError('Invalid quiz data', 400, 'VALIDATION_ERROR');
      }

      const quiz = await service.update(id, parsed.data);

      return reply.code(200).send({
        success: true,
        data: quiz,
      });
    },
  );

  // Release quiz results
  app.post(
    '/quizzes/:id/release-results',
    {
      preHandler: [authenticate, authorize('faculty', 'admin')],
    },
    async (request, reply) => {
      const { id } = request.params as { id: string };

      if (!isUuid(id)) {
        throw new AppError('Invalid quiz ID', 400, 'VALIDATION_ERROR');
      }

      const parsed = releaseQuizResultsSchema.safeParse(request.body);

      if (!parsed.success) {
        throw new AppError(
          'Invalid result release data',
          400,
          'VALIDATION_ERROR',
        );
      }

      const quiz = await service.releaseResults(id, parsed.data);

      return reply.code(200).send({
        success: true,
        data: quiz,
      });
    },
  );

  // Delete a quiz: Admin only
  app.delete(
    '/quizzes/:id',
    {
      preHandler: [authenticate, authorize('admin')],
    },
    async (request, reply) => {
      const { id } = request.params as { id: string };

      if (!isUuid(id)) {
        throw new AppError('Invalid quiz ID', 400, 'VALIDATION_ERROR');
      }

      await service.delete(id);

      return reply.code(200).send({
        success: true,
        message: 'Quiz deleted successfully',
      });
    },
  );
}

function isUuid(value: string): boolean {
  return /^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(
    value,
  );
}
